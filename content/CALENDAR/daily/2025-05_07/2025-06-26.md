# This Terraform configuration grants specific IAM roles to a user in a GCP project.

  

# It's a good practice to define the user's email as a variable.

variable "user_email" {

description = "The email address of the user to grant permissions to."

type = string

default = "filip.naiser@gmail.com"

}

  

# It's also recommended to define your project ID as a variable.

# Replace "your-gcp-project-id" with your actual project ID.

variable "project_id" {

description = "The GCP project ID."

type = string

# provider "google" {} block will implicitly use the project configured in your gcloud or environment,

# but explicitly setting it here is a best practice for clarity and portability.

# default = "your-gcp-project-id"

}

  

# --- Project-Level IAM Bindings ---

# These resources grant broad, admin-level permissions for various services

# at the project level. The "user:" prefix is required to specify a user account.

  

resource "google_project_iam_member" "compute_admin" {

project = var.project_id

role = "roles/compute.admin"

member = "user:${var.user_email}"

}

  

resource "google_project_iam_member" "storage_admin" {

project = var.project_id

role = "roles/storage.admin"

member = "user:${var.user_email}"

}

  

resource "google_project_iam_member" "secret_manager_admin" {

project = var.project_id

role = "roles/secretmanager.admin"

member = "user:${var.user_email}"

}

  

# Granting the Service Account User role at the project level allows the user

# to attach service accounts to resources like GCE instances.

resource "google_project_iam_member" "service_account_user" {

project = var.project_id

role = "roles/iam.serviceAccountUser"

member = "user:${var.user_email}"

}

  
  

# --- Service Account Impersonation ---

# This section grants the user permission to impersonate a specific service account.

  

# First, we need to get the details of the service account you want the user to impersonate.

# Replace 'compute-sa' with the actual account_id of your service account.

# The account_id is the part before the '@' in the service account email.

data "google_service_account" "compute_sa" {

account_id = "compute-sa" # <-- IMPORTANT: Change this to your service account's ID

project = var.project_id

}

  

# Now, grant the user the "Service Account Token Creator" role on that specific service account.

# This is the standard role required for impersonation.

resource "google_service_account_iam_member" "impersonator" {

service_account_id = data.google_service_account.compute_sa.name

role = "roles/iam.serviceAccountTokenCreator"

member = "user:${var.user_email}"

}
